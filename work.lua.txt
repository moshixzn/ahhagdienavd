    "RigType", "HipHeight", "JumpPower", "JumpHeight", "UseJumpPower", "AutoRotate",
    "MaxSlopeAngle", "DisplayDistanceType", "NameDisplayDistance", "HealthDisplayDistance",
    "AutomaticScalingEnabled", "BreakJointsOnDeath", "RequiresNeck", "EvaluateStateMachine",

local function unlockHumanoid()
    local ch = player.Character
    local old = ch and ch:FindFirstChildWhichIsA("Humanoid")
    if not ch or not old then
        return false
    end
    local props = {}
    for _, name in ipairs(K.HUM_COPY) do
        local ok, v = pcall(function() return old[name] end)
        if ok then
            props[name] = v
        end
    end
    local base = old.WalkSpeed
    pcall(function() old:Destroy() end)
    RunService.Heartbeat:Wait()
    local new = Instance.new("Humanoid")
    for name, v in pairs(props) do
        pcall(function() new[name] = v end)
    end
    new.Parent = ch
    RunService.Heartbeat:Wait()
    if new.Parent ~= ch then
        return false
    end
    new.WalkSpeed = base
    pcall(function() new:ChangeState(Enum.HumanoidStateType.Landed) end)
    local cam = Workspace.CurrentCamera
    if cam then
        pcall(function() cam.CameraSubject = new end)
    end
    State.HumReady = true
    return true
end

local function ensureUnlock()
    if State.AntiCheat and not State.HumReady and charReady() then
        return unlockHumanoid()
    end
    return State.HumReady
end

local function net(name)
    return NET and NET:FindFirstChild(name) or nil
end

local function callNet(name, ...)
    local r = net(name)
    if not r then
        return nil
    end
    if r:IsA("RemoteFunction") then
        local ok, res = pcall(r.InvokeServer, r, ...)
        return ok and res or nil
    end
    local ok = pcall(r.FireServer, r, ...)
    return ok or nil
end
local function netNames()
    local out = {}
    if NET then
        for _, r in ipairs(NET:GetChildren()) do
            if r:IsA("RemoteFunction") or r:IsA("RemoteEvent") then
                out[#out + 1] = r.Name
            end
        end
    end
    table.sort(out)
    return out
end

-- // Prompts. HoldDuration is a client side property, so zeroing it right before the
-- fire makes the trigger land in one call on executors whose fireproximityprompt
-- still respects the hold.
local fireprompt = fireproximityprompt or (syn and syn.fireproximityprompt)

local function firePrompt(pp)
    if not fireprompt or not pp or not pp.Enabled then
        return false
    end
    pcall(function() pp.HoldDuration = 0 end)
    return pcall(fireprompt, pp)
end

local function promptInRange(pp)
    local hrp = root()
    local part = pp and pp.Parent
    if not hrp or not part then
        return false
    end
    local pos
    if part:IsA("BasePart") then
        pos = part.Position
    elseif part:IsA("Model") then
        local ok, pv = pcall(function() return part:GetPivot() end)
        pos = ok and pv.Position or nil
    elseif part:IsA("Attachment") then
        pos = part.WorldPosition
    end
    if not pos then
        return false
    end
    local max = pp.MaxActivationDistance > 0 and pp.MaxActivationDistance or 12
    return (hrp.Position - pos).Magnitude <= max + 4
end
local function promptsUnder(inst, words)
    local out = {}
    if not inst then
        return out
    end
    for _, d in ipairs(inst:GetDescendants()) do
        if d:IsA("ProximityPrompt") and d.Enabled then
            local hay = (d.Name .. " " .. d.ActionText .. " " .. d.ObjectText):lower()
            if d.Parent then
                hay = hay .. " " .. d.Parent.Name:lower()
            end
            for _, w in ipairs(words) do
                if hay:find(w, 1, true) then
                    out[#out + 1] = d
                    break
                end
            end
        end
    end
    return out
end

-- // Some endpoints want a table and some want the raw id. Which one worked is
-- remembered per endpoint so the wrong shape is only ever sent once.
M.uidShape = {}

local function askUid(name, uid)
    local shape = M.uidShape[name]
    if shape ~= 2 then
        local res = callNet(name, { Uid = uid })
        if res ~= nil and res ~= false then
            M.uidShape[name] = 1
            return true
        end
        if shape == 1 then
            return false
        end
    end
    local res = callNet(name, uid)
    if res ~= nil and res ~= false then
        M.uidShape[name] = 2
        return true
    end
    return false
end

-- // Areas. The names, their order and the rarity each one holds are read out of
-- Data.Areas, so the dropdown lists whatever this server actually runs instead of a
-- hardcoded guess. Anything the module does not expose is filled in from the field
-- egg snapshot the first time eggs are seen.
A.areaLabel = {}
A.areaRarity = {}
A.areaTierName = {}
A.areaByLabel = {}
A.areaOrder = {}

local function addArea(id, label, rank, tierName)
    if not id or A.areaLabel[id] then
        return
    end
    label = label or id
    A.areaLabel[id] = label
    A.areaRarity[id] = rank or 0
    A.areaTierName[id] = tierName
    A.areaByLabel[label] = id
    A.areaOrder[#A.areaOrder + 1] = id
end

local function sortAreas()
    table.sort(A.areaOrder, function(a, b)
        local ra, rb = A.areaRarity[a] or 0, A.areaRarity[b] or 0
        if ra == rb then
            return tostring(A.areaLabel[a]) < tostring(A.areaLabel[b])
        end
        return ra < rb
    end)
end

local function areaLabels()
    local out = {}
    for _, id in ipairs(A.areaOrder) do
        out[#out + 1] = A.areaLabel[id]
    end
    return out
end
do
    local Data = RS:FindFirstChild("Data")
    local m = Data and Data:FindFirstChild("Areas")
    local ok, mod = pcall(function() return m and require(m) end)
    local dir = ok and type(mod) == "table" and (mod.Directory or mod) or nil
    if type(dir) == "table" then
        for id, info in pairs(dir) do
            if type(id) == "string" and type(info) == "table" then
                local rank, tierName = 0, nil
                if type(info.Rarity) == "table" then
                    rank = tonumber(info.Rarity.RarityNumber) or 0
                    tierName = info.Rarity.DisplayName or info.Rarity._id
                end
                addArea(id, info.DisplayName or info.Name or id, rank, tierName)
            end
        end
    end
    if #A.areaOrder == 0 then
        local folder = Workspace:FindFirstChild("Areas") or Workspace:FindFirstChild("Islands")
        if folder then
            local rank = 0
            for _, a in ipairs(folder:GetChildren()) do
                rank = rank + 1
                addArea(a.Name, a.Name, rank, nil)
            end
        end
    end
    sortAreas()
end

-- // Both field filters are multi select. An empty selection means no restriction, so
-- the sets stay authoritative and nothing has to be reset when the last entry is
-- cleared out of a dropdown.
local function asSet(list)
    local set, n = {}, 0
    if type(list) == "table" then
        for _, v in ipairs(list) do
            if type(v) == "string" and v ~= "" then
                set[v] = true
                n = n + 1
            end
        end
    end
    return set, n
end

local function eggPart(uid)

        return nil

local function eggPos(rec)
    local part = eggPart(rec.Uid)
    if part then
        return part.Position
    end
    local cf = rec.BoundsCFrame or rec.BottomCFrame
    return cf and cf.Position or nil
end

local function isMutated(rec)
    if rec.BaseMutation then
        return true
    end
    return type(rec.Mutations) == "table" and next(rec.Mutations) ~= nil
end

M.eggList = {}
M.eggListAt = 0
M.triedUids = {}
M.areaDirty = false

local function refreshEggs(force)
    if not force and tick() - M.eggListAt < 1.5 then
        return M.eggList
    end
    local snap = callNet(R.EggSnapshot)
    local recs = snap and snap.Records
    if type(recs) ~= "table" then
        return M.eggList
    end
    local list = {}
    for _, rec in pairs(recs) do
        if rec.State == "Slot" and rec.Uid then
            local pos = eggPos(rec)
            if pos then
                local id = rec.AreaId
                if id and not A.areaLabel[id] then
                    addArea(id, id, #A.areaOrder + 1, rarityOf(rec))
                    M.areaDirty = true
                end
                list[#list + 1] = {
                    uid = rec.Uid,
                    area = id,
                    label = id and A.areaLabel[id] or "Unknown",
                    pos = pos,
                    tier = id and A.areaRarity[id] or 0,
                    rarity = id and A.areaTierName[id] or rarityOf(rec),
                    mutated = isMutated(rec),
                    size = rec.BoundsSize and rec.BoundsSize.Magnitude or 3,
                }
            end
        end
    end
    if M.areaDirty then
        M.areaDirty = false
        sortAreas()
        rebuildAreaSet()
        setDropdownValues("AreaFocus", areaLabels())
    end
    M.eggList = list
    M.eggListAt = tick()
    return list
end

local function eligible(e)
    if M.triedUids[e.uid] and tick() - M.triedUids[e.uid] < 6 then
        return false
    end
    if A.areaWantedCount > 0 and not A.areaWanted[e.area] then
        return false
    end
    if A.rarityWantedCount > 0 and not A.rarityWanted[e.rarity] then
        return false
    end
    return true
end

-- The best egg means the highest area rarity first and a mutation on top of that.
-- Distance only breaks the tie, so the farm never settles for a cheap nearby egg.
local function pickEgg()
    local hrp = root()
    if not hrp then
        return nil
    end
    refreshEggs(false)
    local origin = hrp.Position
    local best, bestScore
    for i = 1, #M.eggList do
        local e = M.eggList[i]
        if eligible(e) then
            local d = (e.pos - origin).Magnitude
            local tier = e.tier > 0 and e.tier or ladderOf(e.rarity)
            local score = tier * 1e6 + (e.mutated and 4e5 or 0) - d
            if not bestScore or score > bestScore then
                best, bestScore = e, score
            end
        end
    end
    return best
end
-- // Movement. The route is walked along the floor, not over it: the path is sampled
-- forward in short steps and every step raycasts straight down for the surface, so the
-- root rides hills, ramps and stairs instead of arcing through the air. It is held in
-- the Physics state with the assembly velocity pinned so the humanoid state machine
-- cannot fight the writes and nothing launches when the run ends.
M.travelling = false
M.travelToken = 0

local function cancelTravel()
    M.travelToken = M.travelToken + 1
    M.travelling = false
end

local function pinRoot(hrp, h)
    pcall(function()
        hrp.AssemblyLinearVelocity = Vector3.zero
        hrp.AssemblyAngularVelocity = Vector3.zero
    end)
    if h then
        pcall(function() h:ChangeState(Enum.HumanoidStateType.Physics) end)
    end
end

local function releaseRoot()
    local hrp, h = root(), hum()
    if hrp then
        pcall(function()
            hrp.AssemblyLinearVelocity = Vector3.zero
            hrp.AssemblyAngularVelocity = Vector3.zero
        end)
    end
    if h then
        pcall(function() h:ChangeState(Enum.HumanoidStateType.GettingUp) end)
        pcall(function() h:ChangeState(Enum.HumanoidStateType.Landed) end)
    end
end

M.groundParams = RaycastParams.new()
M.groundParams.FilterType = Enum.RaycastFilterType.Exclude
M.groundParams.IgnoreWater = true

-- Field eggs are solid, so a down ray that lands on one stands the root on its shell and
-- the run climbs into the air. The egg folder and every character are excluded outright.
-- Nothing else is excluded by name: a folder called after the world would take the floor
-- with it and the whole route would be flying.
local function groundSkip()
    local skip = {}
    if player.Character then
        skip[#skip + 1] = player.Character
    end
    for _, p in ipairs(players:GetPlayers()) do
        if p ~= player and p.Character then
            skip[#skip + 1] = p.Character
        end
    end
    local eggs = slots()
    if eggs then
        skip[#skip + 1] = eggs
    end
    return skip
end

local function refreshGroundFilter()
    M.groundParams.FilterDescendantsInstances = groundSkip()
end

K.PROBE_LOW = 7
K.PROBE_HIGH = 160
K.PROBE_DOWN = 420
K.STEP_MAX = 6
K.TRAVEL_SPEED = 1000

-- Pets cannot be excluded by folder name because their container is not in any dump, but
-- an animated rig always carries a Humanoid or an AnimationController and a floor never
-- does. That is what separates something to stand on from something to walk through.
local function notFloor(inst)
    local node = inst
    for _ = 1, 6 do
        if not node or node == Workspace then
            return false
        end
        if node:IsA("Model")
            and (node:FindFirstChildWhichIsA("Humanoid") or node:FindFirstChildWhichIsA("AnimationController")) then
            return true
        end
        node = node.Parent
    end
    return false
end

local function castFloor(x, z, top, drop)
    local y = top
    for _ = 1, 4 do
        local res = Workspace:Raycast(Vector3.new(x, y, z), Vector3.new(0, -(y - (top - drop)), 0), M.groundParams)
        if not res then
            return nil
        end
        if not notFloor(res.Instance) then
            return res.Position.Y
        end
        y = res.Position.Y - 0.6
        if y <= top - drop then
            return nil
        end
    end
    return nil
end

-- Two probes, the low one first. Starting high and taking the first hit is what put the run
-- on top of roofs and egg shells, so the surface just under the feet always wins and the
-- tall probe is only there to climb a ledge when there is no floor below at all.
local function floorY(x, z, near)
    local y = castFloor(x, z, near + K.PROBE_LOW, K.PROBE_LOW + K.PROBE_DOWN)
    if y then
        return y
    end
    y = castFloor(x, z, near + K.PROBE_HIGH, K.PROBE_HIGH + K.PROBE_DOWN)
    if y and y > near + K.PROBE_LOW then
        return y
    end
    return nil
end

local function rootRise()
    local hrp, h = root(), hum()
    local half = hrp and hrp.Size.Y * 0.5 or 1
    local hip = 2
    if h then
        local ok, v = pcall(function() return h.HipHeight end)
        if ok and type(v) == "number" and v > 0 then
            hip = v
        end
    end
    return half + hip
end
-- // Routing. A straight line between two points on this map crosses open space, and the
-- run used to drop onto whatever was underneath and come back along the outside. A line is
-- only accepted when every sample along it lands on floor and no sample drops off a ledge,
-- so a route that leaves the island is rejected before a single frame of it is walked.
M.PathService = game:GetService("PathfindingService")
K.ROUTE_SAMPLE = 14
K.ROUTE_DROP = 26

local function walkable(a, b, floor)
    local flatVec = Vector3.new(b.X - a.X, 0, b.Z - a.Z)
    local dist = flatVec.Magnitude
    if dist < 1 then
        return true, floor
    end
    local dir = flatVec.Unit
    local steps = math.ceil(dist / K.ROUTE_SAMPLE)
    local y = floor
    for i = 1, steps do
        local p = a + dir * math.min(dist, i * K.ROUTE_SAMPLE)
        local g = floorY(p.X, p.Z, y)
        if not g or math.abs(g - y) > K.ROUTE_DROP then
            return false, y
        end
        y = g
    end
    return true, y
end

local function chainWalkable(from, points)
    local y = floorY(from.X, from.Z, from.Y)
    if not y then
        return false
    end
    local prev = from
    for i = 1, #points do
        local ok, endY = walkable(prev, points[i], y)
        if not ok then
            return false
        end
        prev, y = points[i], endY
    end
    return true
end
-- The middle of the map is where every route is bent through when the straight line is not
-- walkable. Stands sit in the hub, and the plot ring averages out to the same place, so
-- either one gives a centre that is real geometry instead of a guessed coordinate.
M.hubCache, M.hubTried = nil, false

local function centroidOf(scope, cap)
    if not scope then
        return nil
    end
    local sum, n = Vector3.zero, 0
    for _, d in ipairs(scope:GetDescendants()) do
        if d:IsA("BasePart") then
            sum = sum + d.Position
            n = n + 1
            if n >= cap then
                break
            end
        end
    end
    if n == 0 then
        return nil
    end
    return sum / n
end

local function hubPoint()
    if M.hubCache or M.hubTried then
        return M.hubCache
    end
    M.hubTried = true
    local mid = centroidOf(Workspace:FindFirstChild("Stands"), 60)
    if not mid then
        local plots = Workspace:FindFirstChild("Plots")
        if plots then
            local sum, n = Vector3.zero, 0
            for _, p in ipairs(plots:GetChildren()) do
                if p:IsA("Model") then
                    local ok, pv = pcall(function() return p:GetPivot() end)
                    if ok and pv then
                        sum = sum + pv.Position
                        n = n + 1
                    end
                end
            end
            if n > 0 then
                mid = sum / n
            end
        end
    end
    if mid then
        local g = floorY(mid.X, mid.Z, mid.Y) or floorY(mid.X, mid.Z, mid.Y + 200)
        if g then
            M.hubCache = Vector3.new(mid.X, g, mid.Z)
        end
    end
    return M.hubCache
end
local function simplify(points)
    local out = {}
    for i = 1, #points do
        local p = points[i]
        local last = out[#out]
        if not last or Vector3.new(p.X - last.X, 0, p.Z - last.Z).Magnitude > 3 then
            out[#out + 1] = p
        end
    end
    if #out < 3 then
        return out
    end
    local keep = {}
    for i = 2, #out - 1 do
        local a = Vector3.new(out[i].X - out[i - 1].X, 0, out[i].Z - out[i - 1].Z)
        local b = Vector3.new(out[i + 1].X - out[i].X, 0, out[i + 1].Z - out[i].Z)
        if a.Magnitude > 0.1 and b.Magnitude > 0.1 and a.Unit:Dot(b.Unit) < 0.995 then
            keep[#keep + 1] = out[i]
        end
    end
    keep[#keep + 1] = out[#out]
    return keep
end

local function pathRoute(from, to)
    local ok, path = pcall(function()
        return M.PathService:CreatePath({
            AgentRadius = 3,
            AgentHeight = 6,
            AgentCanJump = true,
            AgentCanClimb = false,
            WaypointSpacing = 24,
        })
    end)
    if not ok or not path then
        return nil
    end
    local ran = pcall(function() path:ComputeAsync(from, to) end)
    local status, wps
    pcall(function() status = path.Status end)
    if ran and status == Enum.PathStatus.Success then
        pcall(function() wps = path:GetWaypoints() end)
    end
    pcall(function() path:Destroy() end)
    if type(wps) ~= "table" or #wps < 2 then
        return nil
    end
    local pts = {}
    for i = 2, #wps do
        pts[#pts + 1] = wps[i].Position
    end
    pts[#pts] = to
    return simplify(pts)
end
K.SIDE_OFFSETS = { 30, -30, 70, -70, 140, -140, 240, -240 }

-- Cheapest first: the straight line, then a bend through the middle of the map, then a
-- sideways bend, and the navmesh only when geometry beats all of them. Every option that
-- gets returned has been probed end to end, so nothing that leaves the map is handed back.
local function routeTo(from, to)
    if chainWalkable(from, { to }) then
        return { to }
    end
    local mid = hubPoint()
    if mid and chainWalkable(from, { mid, to }) then
        return { mid, to }
    end
    local flatVec = Vector3.new(to.X - from.X, 0, to.Z - from.Z)
    if flatVec.Magnitude > 1 then
        local dir = flatVec.Unit
        local side = Vector3.new(-dir.Z, 0, dir.X)
        local centre = from + flatVec * 0.5
        for i = 1, #K.SIDE_OFFSETS do
            local w = centre + side * K.SIDE_OFFSETS[i]
            local g = floorY(w.X, w.Z, centre.Y)
            if g then
                local wp = Vector3.new(w.X, g, w.Z)
                if chainWalkable(from, { wp, to }) then
                    return { wp, to }
                end
            end
        end
    end
    local viaMesh = pathRoute(from, to)
    if viaMesh and #viaMesh > 0 then
        return viaMesh
    end
    return { to }
end
local function travelLeg(targetPos, reachDist, keepGoing, token)
    local hrp = root()
    if not hrp then
        return false
    end
    local start = hrp.Position
    local flatVec = Vector3.new(targetPos.X - start.X, 0, targetPos.Z - start.Z)
    local flat = flatVec.Magnitude
    if flat <= 0.5 then
        return true
    end

    local dir = flatVec.Unit
    local yaw = CFrame.lookAt(Vector3.zero, dir).Rotation
    local rise = rootRise()
    local y = (floorY(start.X, start.Z, start.Y - rise) or (start.Y - rise)) + rise
    local moved = 0
    local deadline = tick() + flat / K.TRAVEL_SPEED + 10
    local done = false

    while alive() and M.travelToken == token do
        if keepGoing and not keepGoing() then
            break
        end
        local dt = RunService.Heartbeat:Wait()
        local r, h = root(), hum()
        if not r or not r.Parent or not h or h.Health <= 0 then
            break
        end
        if tick() > deadline then
            break
        end
        local frameStep = math.min(K.TRAVEL_SPEED * math.min(dt, 0.1), flat - moved)
        local subs = math.max(1, math.ceil(frameStep / K.STEP_MAX))
        local per = frameStep / subs
        for _ = 1, subs do
            moved = math.min(flat, moved + per)
            local p = start + dir * moved
            local g = floorY(p.X, p.Z, y - rise)
            if g then
                y = y + math.clamp((g + rise) - y, -K.STEP_MAX * 4, K.STEP_MAX * 4)
            end
            if moved >= flat then
                break
            end
        end
        local p = start + dir * moved
        pcall(function() r.CFrame = CFrame.new(p.X, y, p.Z) * yaw end)
        pinRoot(r, h)
        if moved >= flat - 0.01 then
            done = true
            break
        end
    end

    if M.travelToken ~= token then
        return false
    end
    if done then
        return true
    end
    local r = root()
    if not r then
        return false
    end
    return Vector3.new(targetPos.X - r.Position.X, 0, targetPos.Z - r.Position.Z).Magnitude <= (reachDist or 8)
end

local function travelTo(targetPos, reachDist, keepGoing)
    local hrp = root()
    if not hrp or not targetPos then
        return false
    end
    reachDist = reachDist or 8
    if Vector3.new(targetPos.X - hrp.Position.X, 0, targetPos.Z - hrp.Position.Z).Magnitude <= reachDist then
        return true
    end

    cancelTravel()
    refreshGroundFilter()
    local token = M.travelToken
    M.travelling = true

    local legs = routeTo(hrp.Position, targetPos)
    local ok = true
    for i = 1, #legs do
        if not travelLeg(legs[i], i == #legs and reachDist or 4, keepGoing, token) then
            ok = false
            break
        end
    end

    if M.travelToken ~= token then
        return false
    end
    M.travelling = false
    releaseRoot()
    local r = root()
    if not r then
        return false
    end
    local left = Vector3.new(targetPos.X - r.Position.X, 0, targetPos.Z - r.Position.Z).Magnitude
    return ok and left <= reachDist + 6
end
-- // Turning the farm off drops the character through the map. It holds just above the
-- destroy height for five seconds, then sinks past it so the server takes the rig and
-- respawns it. If nothing comes back within twenty seconds the rig wipe endpoint is the
-- fallback and the warning says so.
M.voidBusy = false

local function backOnMap(floor)
    if not charReady() then
        return false
    end
    local r = root()
    return r ~= nil and r.Parent ~= nil and r.Position.Y > floor
end

local function voidDrop()
    if M.voidBusy then
        return false
    end
    local hrp = root()
    if not hrp then
        return false
    end
    M.voidBusy = true
    cancelTravel()
    local killY = -500
    local okY, v = pcall(function() return Workspace.FallenPartsDestroyHeight end)
    if okY and type(v) == "number" then
        killY = v
    end
    local safe = killY + 60
    local p = hrp.Position
    local h = hum()
    if h then
        pcall(function() h:ChangeState(Enum.HumanoidStateType.Physics) end)
    end
    local hold = tick() + 5
    while alive() and tick() < hold do
        local r = root()
        if not r or not r.Parent then
            break
        end
        pcall(function()
            r.CFrame = CFrame.new(p.X, safe, p.Z)
            r.AssemblyLinearVelocity = Vector3.zero
            r.AssemblyAngularVelocity = Vector3.zero
        end)
        RunService.Heartbeat:Wait()
    end
    local r = root()
    if r and r.Parent then
        pcall(function()
            r.CFrame = CFrame.new(p.X, killY - 250, p.Z)
            r.AssemblyLinearVelocity = Vector3.new(0, -280, 0)
        end)
    end
    local watch = killY + 200
    local deadline = tick() + 20
    local back = false
    while alive() and tick() < deadline do
        if backOnMap(watch) then
            back = true
            break
        end
        task.wait(0.2)
    end
    if not back and alive() then
        warnToast("Using Fallback..", 5)
        callNet(R.RigWipe)
        local retry = tick() + 12
        while alive() and tick() < retry do
            if backOnMap(watch) then
                back = true
                break
            end
            task.wait(0.2)
        end
    end
    M.voidBusy = false
    return back
end

-- Placement is asked for in plot space, so the pen is wherever the plot origin puts
-- the middle of the grid. A model actually named after the pen wins if the plot has one.
-- The descendant sweep is kept out of the run home by caching what it found.
M.penCache = nil

local function penPivot()
    if M.penCache then
        return M.penCache
    end
    local p = myPlot()
    if p then
        for _, d in ipairs(p:GetDescendants()) do
            if d.Name:lower():find("pen", 1, true) then
                if d:IsA("BasePart") then
                    M.penCache = d.CFrame
                    return M.penCache
                end
                if d:IsA("Model") then
                    local ok, pv = pcall(function() return d:GetPivot() end)
                    if ok and pv then
                        M.penCache = pv
                        return M.penCache
                    end
                end
            end
        end
    end
    local base = plotPivot()
    if base then
        M.penCache = base * CFrame.new(0, 0, 15)
        return M.penCache
    end
    return nil
end

local function myRecords()
    local out = {}
    local live = callNet(R.EggLive)
    if type(live) == "table" then
        for _, entry in pairs(live) do
            if type(entry) == "table" and entry.OwnerUserId == player.UserId
                and type(entry.Records) == "table" then
                for _, rec in pairs(entry.Records) do
                    out[#out + 1] = rec
                end
            end
        end
    end
    return out
end

local function usedSpots()
    local used = {}
    for _, rec in ipairs(myRecords()) do
        if rec.Placement and rec.Placement.LocalCFrame then
            used[#used + 1] = rec.Placement.LocalCFrame.Position
        end
    end
    return used
end
-- LocalCFrame is plot relative, so the rows are built back to front and left to
-- right: filling them in that order is what keeps the plot in tidy lines.
K.placeGrid = {}
for z = 0, 30, 5 do
    for x = -20, 20, 5 do
        K.placeGrid[#K.placeGrid + 1] = Vector2.new(x, z)
    end
end

-- The live snapshot lags a moment behind a successful place, so cells filled this
-- session are remembered or the next egg would aim at the same one.
M.claimedCells = {}

local function cellKey(g)
    return g.X .. ":" .. g.Y
end

local function freeCell(g, used)
    if M.claimedCells[cellKey(g)] then
        return false
    end
    for _, p in ipairs(used) do
        if math.abs(p.X - g.X) < 3.5 and math.abs(p.Z - g.Y) < 3.5 then
            return false
        end
    end
    return true
end

-- The place remote is a RemoteFunction that answers true only when the payload is the
-- shape it wants, so the first successful shape is remembered and reused after that.
M.placeShape = 0
M.placeProbes = 0

local function tryPlace(uid, cf)
    local r = net(R.EggPlace)
    if not r or not r:IsA("RemoteFunction") then
        return false
    end
    local function shot(n)
        if n == 1 then
            return r:InvokeServer({ Uid = uid, LocalCFrame = cf })
        elseif n == 2 then
            return r:InvokeServer({ Uid = uid, CFrame = cf })
        elseif n == 3 then
            return r:InvokeServer(uid, cf)
        end
        return r:InvokeServer({ Uid = uid, Placement = { LocalCFrame = cf } })
    end
    if M.placeShape > 0 then
        local ok, res = pcall(shot, M.placeShape)
        return ok and res == true
    end
    for n = 1, 4 do
        local ok, res = pcall(shot, n)
        if ok and res == true then
            M.placeShape = n
            return true
        end
    end
    M.placeProbes = M.placeProbes + 1
    if M.placeProbes >= 3 then
        M.placeShape = 1
    end
    return false
end

local function placeEgg(uid, maxTries)
    maxTries = maxTries or 14
    local used = usedSpots()
    local tries = 0
    for _, g in ipairs(K.placeGrid) do
        if freeCell(g, used) then
            tries = tries + 1
            if tryPlace(uid, CFrame.new(g.X, 0, g.Y)) then
                M.claimedCells[cellKey(g)] = true
                return true
            end
            if tries >= maxTries then
                break
            end
        end
    end
    return false
end

-- AskFieldEggCarry claims the egg, AskWearTool is what actually puts it in the hand. The
-- carry answer is the only thing worth waiting for: a creature is on the egg the moment it
-- is claimed, so the tool call is fired off to the side and the run home starts on the
-- same frame instead of a third of a second later.
local function holdingEgg()
    local ch = player.Character
    return ch ~= nil and ch:FindFirstChildWhichIsA("Tool") ~= nil
end

local function grabEgg(uid)
    if callNet(R.EggCarry, { Uid = uid }) ~= true then
        return false
    end
    task.spawn(function()
        for _ = 1, 3 do
            if holdingEgg() then
                return
            end
            askUid(R.WearTool, uid)
            task.wait(0.15)
        end
    end)
    return true
end
-- The egg model leaves the client folder the moment somebody claims it, so an
-- in-flight trip is dropped as soon as its target is gone.
local function eggAlive(uid)
    local folder = slots()
    if not folder then
        return true
    end
    return folder:FindFirstChild(uid) ~= nil
end

K.GRAB_TOLERANCE = 8

local function inRange(pos, reach)
    local hrp = root()
    if not hrp or not pos then
        return false
    end
    local a = Vector3.new(hrp.Position.X, 0, hrp.Position.Z)
    local b = Vector3.new(pos.X, 0, pos.Z)
    return (a - b).Magnitude <= (reach or 12) + K.GRAB_TOLERANCE
end

M.carryingUid = nil

-- The snapshot is a second and a half old at worst and the record bounds are not always
-- where the shell ended up, so the model in AreaEggSlotsClient is asked again right before
-- the run starts and once more after it lands.
local function livePos(e)
    local part = eggPart(e.uid)
    if part then
        e.pos = part.Position
    end
    return e.pos
end

local function autoStealOn()
    return alive() and State.AutoSteal
end

-- The run home is base first, then the pen, then the place call. Placement is always
-- asked for from the plot, never from wherever the egg was picked up.
local function deliver(uid)
    local base = plotPivot()
    if base and not inRange(base.Position, 26) then
        travelTo(base.Position, 14, autoStealOn)
    end
    local pen = penPivot()
    if pen and not inRange(pen.Position, 12) then
        travelTo(pen.Position, 8, autoStealOn)
    end
    for _ = 1, 2 do
        if placeEgg(uid) then
            return true
        end
        task.wait(0.2)
    end
    for _, pp in ipairs(promptsUnder(myPlot(), { "place", "drop", "put" })) do
        if promptInRange(pp) and firePrompt(pp) then
            task.wait(0.15)
            return true
        end
    end
    return false
end

local function goHome(reach)
    local pv = plotPivot()
    if not pv then
        return false
    end
    if inRange(pv.Position, reach or 20) then
        return true
    end
    return travelTo(pv.Position, 12)
end
-- // Auto Steal. One egg at a time: pick the best one the filters allow, run to it on a
-- route that never leaves the map, claim it and leave on the same frame the claim answers,
-- then place it in the pen. Nothing here needs a second toggle switched on first.
spawnLoop(function()
    while alive() do
        if State.AutoSteal and charReady() then
            local ok, err = pcall(function()
                if M.carryingUid then
                    if deliver(M.carryingUid) then
                        M.carryingUid = nil
                    end
                    return
                end
                local e = pickEgg()
                if not e then
                    return
                end
                M.triedUids[e.uid] = tick()
                local reach = math.max(9, e.size * 0.6 + 7)
                local keep = function()
                    return autoStealOn() and eggAlive(e.uid)
                end
                local target = livePos(e)
                if not inRange(target, reach) then
                    travelTo(target, reach, keep)
                    if not autoStealOn() then
                        return
                    end
                    local fresh = livePos(e)
                    if not inRange(fresh, reach) then
                        travelTo(fresh, reach, keep)
                        if not autoStealOn() then
                            return
                        end
                    end
                end
                local got = grabEgg(e.uid)
                if not got then
                    task.wait(0.1)
                    got = grabEgg(e.uid)
                end
                if got then
                    M.carryingUid = e.uid
                    if deliver(e.uid) then
                        M.carryingUid = nil
                    end
                end
            end)
            if not ok then
                warn("[Shard] steal: " .. tostring(err))
                task.wait(0.5)
            end
        end
        RunService.Heartbeat:Wait()
    end
end)
-- // Replacing the Humanoid leaves the default control script writing to a dead
-- object, so WASD and Space stop responding. These re-drive both from raw input.
K.MOVE_KEYS = {
    [Enum.KeyCode.W] = Vector3.new(0, 0, -1),
    [Enum.KeyCode.S] = Vector3.new(0, 0, 1),
    [Enum.KeyCode.A] = Vector3.new(-1, 0, 0),
    [Enum.KeyCode.D] = Vector3.new(1, 0, 0),
}
M.held = {}

track(UIS.InputBegan:Connect(function(input, gp)
    if gp then
        return
    end
    if K.MOVE_KEYS[input.KeyCode] then
        M.held[input.KeyCode] = true
    end
end))

track(UIS.InputEnded:Connect(function(input)
    if K.MOVE_KEYS[input.KeyCode] then
        M.held[input.KeyCode] = nil
    end
end))

track(UIS.JumpRequest:Connect(function()
    if not State.HumReady or M.travelling then
        return
    end
    local h = hum()
    if h and h.Health > 0 then
        pcall(function() h:ChangeState(Enum.HumanoidStateType.Jumping) end)
    end
end))

local function manualDir()
    local cam = Workspace.CurrentCamera
    if not cam then
        return Vector3.zero
    end
    local raw = Vector3.zero
    for key, offset in pairs(K.MOVE_KEYS) do
        if M.held[key] then
            raw = raw + offset
        end
    end
    if raw.Magnitude < 0.01 then
        return Vector3.zero
    end
    local cf = cam.CFrame
    local dir = cf.RightVector * raw.X - cf.LookVector * raw.Z
    return Vector3.new(dir.X, 0, dir.Z)
end
spawnLoop(function()
    while alive() do
        RunService.Heartbeat:Wait()
        if State.HumReady and not M.travelling and not State.AutoTreadmill then
            local h = hum()
            if h and h.Health > 0 then
                local dir = manualDir()
                if dir.Magnitude > 0.01 then
                    h:Move(dir.Unit, false)
                end
            end
        end
    end
end)

-- // Hatching. AskHatch starts one, AskFinishHatch closes it out and AskSkipGrowth
-- pushes an egg that is still growing straight to ready.
local function hatchAll(walk)
    local recs = myRecords()
    local n = 0
    for _, rec in ipairs(recs) do
        local uid = rec.Uid
        if uid then
            if askUid(R.HatchFinish, uid) then
                n = n + 1
            elseif askUid(R.Hatch, uid) then
                n = n + 1
            elseif askUid(R.SkipGrowth, uid) and askUid(R.HatchFinish, uid) then
                n = n + 1
            end
            task.wait(0.06)
        end
    end
    if n == 0 then
        if walk then
            goHome(20)
        end
        for _, pp in ipairs(promptsUnder(myPlot(), { "hatch", "open" })) do
            if promptInRange(pp) and firePrompt(pp) then
                n = n + 1
                task.wait(0.08)
            end
        end
    end
    return n
end

spawnLoop(function()
    while alive() do
        if (State.AutoHatch or State.HatchOnce) and charReady() and not M.travelling then
            pcall(hatchAll, false)
            if State.HatchOnce then
                State.HatchOnce = false
            end
        end
        task.wait(3)
    end
end)

M.espFolder = nil
M.espTags = {}

local function espParent()
    if M.espFolder and M.espFolder.Parent then
        return M.espFolder
    end
    M.espFolder = Instance.new("Folder")
    M.espFolder.Name = "ShardEggESP"
    M.espFolder.Parent = (gethui and gethui()) or game:GetService("CoreGui")
    return M.espFolder
end
local function clearESP()
    for uid, tag in pairs(M.espTags) do
        pcall(function() tag:Destroy() end)
        M.espTags[uid] = nil
    end
    if M.espFolder then
        pcall(function() M.espFolder:Destroy() end)
        M.espFolder = nil
    end
end

local function makeTag(e)
    local mdl = eggModel(e.uid)
    local part = eggPart(e.uid)
    if not mdl or not part then
        return nil
    end
    local colour = A.rarityColor[e.rarity] or Color3.fromRGB(255, 255, 255)
    local box = Instance.new("Folder")
    box.Name = e.uid

    local hl = Instance.new("Highlight")
    hl.Adornee = mdl
    hl.FillColor = colour
    hl.OutlineColor = e.mutated and Color3.fromRGB(255, 150, 60) or colour
    hl.FillTransparency = 0.62
    hl.OutlineTransparency = 0
    hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
    hl.Parent = box

    local bb = Instance.new("BillboardGui")
    bb.Adornee = part
    bb.Size = UDim2.fromOffset(190, 26)
    bb.StudsOffsetWorldSpace = Vector3.new(0, 3.2, 0)
    bb.AlwaysOnTop = true
    bb.MaxDistance = 100000
    bb.Parent = box

    local lbl = Instance.new("TextLabel")
    lbl.Size = UDim2.fromScale(1, 1)
    lbl.BackgroundTransparency = 1
    lbl.Font = Enum.Font.GothamBold
    lbl.TextSize = 13
    lbl.TextColor3 = colour
    lbl.TextStrokeTransparency = 0.35
    lbl.Text = (e.rarity or e.label or "Egg") .. (e.mutated and "  MUT" or "")
    lbl.Parent = bb

    box.Parent = espParent()
    return box
end

-- // FPS Boost. Rendering drops to the cheapest settings the client accepts and every
-- live effect in the world is switched off. Pets are what fill this map, so they go too:
-- with a busy lobby they are most of what the GPU is asked to draw.
M.Lighting = game:GetService("Lighting")

M.fpsSaved = nil
M.hiddenParts = {}
M.hiddenGuis = {}
M.killedFx = {}

local function isCharacterPart(inst)
    for _, p in ipairs(players:GetPlayers()) do
        local ch = p.Character
        if ch and (inst == ch or inst:IsDescendantOf(ch)) then
            return true
        end
    end
    return false
end

local function petish(inst)
    local node = inst
    for _ = 1, 8 do
        if not node or node == Workspace then
            return false
        end
        local low = node.Name:lower()
        if low:find("pet") or low:find("pen") then
            return true
        end
        node = node.Parent
    end
    return false
end

local function petHosts()
    local out = {}
    for _, d in ipairs(Workspace:GetChildren()) do
        local low = d.Name:lower()
        if low:find("pet") or low:find("pen") then
            out[#out + 1] = d
        end
    end
    local plots = Workspace:FindFirstChild("Plots")
    if plots then
        out[#out + 1] = plots
    end
    return out
end
local function hidePart(part)
    if M.hiddenParts[part] ~= nil then
        return
    end
    M.hiddenParts[part] = part.LocalTransparencyModifier
    pcall(function() part.LocalTransparencyModifier = 1 end)
end

local function hideGui(gui)
    if M.hiddenGuis[gui] ~= nil then
        return
    end
    M.hiddenGuis[gui] = gui.Enabled
    pcall(function() gui.Enabled = false end)
end

local function killFx(inst)
    if M.killedFx[inst] ~= nil then
        return
    end
    if inst:IsA("ParticleEmitter") or inst:IsA("Trail") or inst:IsA("Beam")
        or inst:IsA("Smoke") or inst:IsA("Fire") or inst:IsA("Sparkles")
        or inst:IsA("PostEffect") then
        M.killedFx[inst] = inst.Enabled
        pcall(function() inst.Enabled = false end)
    end
end

local function sweepPets(list)
    for i = 1, #list do
        local d = list[i]
        killFx(d)
        if not isCharacterPart(d) and petish(d) then
            if d:IsA("BasePart") then
                hidePart(d)
            elseif d:IsA("BillboardGui") then
                hideGui(d)
            end
        end
    end
end
local function fpsOn()
    local terrain = Workspace:FindFirstChildWhichIsA("Terrain")
    if not M.fpsSaved then
        M.fpsSaved = {
            shadows = M.Lighting.GlobalShadows,
            diffuse = M.Lighting.EnvironmentDiffuseScale,
            specular = M.Lighting.EnvironmentSpecularScale,
            fogEnd = M.Lighting.FogEnd,
            terrain = terrain,
        }
        pcall(function() M.fpsSaved.quality = settings().Rendering.QualityLevel end)
        if terrain then
            M.fpsSaved.waveSize = terrain.WaterWaveSize
            M.fpsSaved.waveSpeed = terrain.WaterWaveSpeed
            M.fpsSaved.reflect = terrain.WaterReflectance
        end
    end
    pcall(function() settings().Rendering.QualityLevel = Enum.QualityLevel.Level01 end)
    pcall(function() M.Lighting.GlobalShadows = false end)
    pcall(function() M.Lighting.EnvironmentDiffuseScale = 0 end)
    pcall(function() M.Lighting.EnvironmentSpecularScale = 0 end)
    pcall(function() M.Lighting.FogEnd = 100000 end)
    if terrain then
        pcall(function()
            terrain.WaterWaveSize = 0
            terrain.WaterWaveSpeed = 0
            terrain.WaterReflectance = 0
        end)
    end
    for _, e in ipairs(M.Lighting:GetChildren()) do
        killFx(e)
    end
    sweepPets(Workspace:GetDescendants())
end
local function fpsOff()
    for part, v in pairs(M.hiddenParts) do
        pcall(function() part.LocalTransparencyModifier = v end)
        M.hiddenParts[part] = nil
    end
    for gui, v in pairs(M.hiddenGuis) do
        pcall(function() gui.Enabled = v end)
        M.hiddenGuis[gui] = nil
    end
    for inst, v in pairs(M.killedFx) do
        pcall(function() inst.Enabled = v end)
        M.killedFx[inst] = nil
    end
    if not M.fpsSaved then
        return
    end
    pcall(function() M.Lighting.GlobalShadows = M.fpsSaved.shadows end)
    pcall(function() M.Lighting.EnvironmentDiffuseScale = M.fpsSaved.diffuse end)
    pcall(function() M.Lighting.EnvironmentSpecularScale = M.fpsSaved.specular end)
    pcall(function() M.Lighting.FogEnd = M.fpsSaved.fogEnd end)
    if M.fpsSaved.quality then
        pcall(function() settings().Rendering.QualityLevel = M.fpsSaved.quality end)
    end
    local terrain = M.fpsSaved.terrain
    if terrain and M.fpsSaved.waveSize then
        pcall(function()
            terrain.WaterWaveSize = M.fpsSaved.waveSize
            terrain.WaterWaveSpeed = M.fpsSaved.waveSpeed
            terrain.WaterReflectance = M.fpsSaved.reflect
        end)
    end
    M.fpsSaved = nil
end

spawnLoop(function()
    while alive() do
        task.wait(3)
        if State.FpsBoost then
            pcall(function()
                for _, host in ipairs(petHosts()) do
                    sweepPets(host:GetDescendants())
                end
            end)
        end
    end
end)

local function goSellStand()
    local cf = sellStandCFrame()
    if not cf then
        return false
    end
    if inRange(cf.Position, 14) then
        return true
    end
    return travelTo(cf.Position, 8)
end
local function sellOnePet(uid)
    if askUid(R.PetSale, uid) then
        return true
    end
    return callNet(R.SellPet, { Uid = uid }) ~= nil
end

local function sellPets(walk)
    local want = raritySet(State.SellRarities)
    if not next(want) then
        if walk then
            goSellStand()
        end
        fireSellPrompt("SellAll")
        callNet(R.SellEveryPet)
        callNet(R.SatchelSale)
        return "Sold every pet"
    end
    if walk then
        goSellStand()
    end
    local n = 0
    for _, rec in ipairs(petSnapshot()) do
        local r = rarityOf(rec)
        if type(rec) == "table" and rec.Uid and r and want[r] and not rec.Equipped then
            if sellOnePet(rec.Uid) then
                n = n + 1
            end
            task.wait(0.05)
        end
    end
    return "Sold " .. n .. " pets"
end

-- Eggs are not sold at the stand in this game, they hatch into pets. Auto Sell Eggs
-- finishes the eggs on the plot and then sells the pets that came out of them.

    local n = hatchAll(walk)

    if not next(want) then
        fireSellPrompt("SellAll")
        callNet(R.SellEveryPet)
        return "Hatched " .. n .. " and sold every pet"

    local sold = 0
    for _, rec in ipairs(petSnapshot()) do
        local r = rarityOf(rec)
        if type(rec) == "table" and rec.Uid and r and want[r] and not rec.Equipped then
            if sellOnePet(rec.Uid) then
                sold = sold + 1
            end
            task.wait(0.05)
        end
    end
    return "Hatched " .. n .. " and sold " .. sold
end

-- // Treadmill. Name matching alone kept landing on the upgrade board, because the board is
-- a descendant named after the treadmill and it comes first in the sweep. Anything that
-- reads like signage is thrown out, the candidate nearest the plot wins, and inside it the
-- seat is preferred over the belt so the character actually gets on the thing.
K.TREAD_AVOID = {
    "upgrade", "tier", "level", "board", "sign", "billboard", "gui", "shop", "buy",
    "display", "screen", "label", "text", "icon", "price", "cost", "info", "menu",
}

local function treadAvoided(name)
    local low = name:lower()
    for i = 1, #K.TREAD_AVOID do
        if low:find(K.TREAD_AVOID[i], 1, true) then
            return true
        end
    end
    return false
end

local function treadCollect(scope, out)
    if not scope then
        return out
    end
    if (scope:IsA("Model") or scope:IsA("BasePart"))
        and scope.Name:lower():find("treadmill", 1, true) and not treadAvoided(scope.Name) then
        out[#out + 1] = scope
    end
    for _, d in ipairs(scope:GetDescendants()) do
        if (d:IsA("Model") or d:IsA("BasePart"))
            and d.Name:lower():find("treadmill", 1, true) and not treadAvoided(d.Name) then
            out[#out + 1] = d
        end
    end
    return out
end

local function anchorOf(inst)
    if inst:IsA("BasePart") then
        return inst.Position
    end
    local ok, pv = pcall(function() return inst:GetPivot() end)
    return ok and pv and pv.Position or nil
end
M.treadCache = nil

local function treadmill()
    if M.treadCache and M.treadCache.Parent then
        return M.treadCache
    end
    local out = {}
    treadCollect(myPlot(), out)
    if #out == 0 then
        treadCollect(Workspace:FindFirstChild("Treadmills"), out)
    end
    if #out == 0 then
        treadCollect(Workspace:FindFirstChild("Stands"), out)
    end
    if #out == 0 then
        for _, c in ipairs(Workspace:GetChildren()) do
            if c.Name:lower():find("tread", 1, true) then
                treadCollect(c, out)
            end
        end
    end
    if #out == 0 then
        return nil
    end
    local base = plotPivot()
    local hrp = root()
    local origin = base and base.Position or (hrp and hrp.Position)
    local best, bestScore
    for i = 1, #out do
        local a = anchorOf(out[i])
        if a then
            local d = origin and (a - origin).Magnitude or 0
            local score = (out[i]:IsA("Model") and 40 or 0) - d
            if not bestScore or score > bestScore then
                best, bestScore = out[i], score
            end
        end
    end
    M.treadCache = best
    return M.treadCache
end

local function treadSeat()
    local t = treadmill()
    if not t then
        return nil
    end
    if t:IsA("Seat") or t:IsA("VehicleSeat") then
        return t
    end
    if t:IsA("BasePart") then
        return nil
    end
    for _, d in ipairs(t:GetDescendants()) do
        if d:IsA("Seat") or d:IsA("VehicleSeat") then
            return d
        end
    end
    return nil
end

K.BELT_HINTS = { "belt", "tread", "walk", "run", "platform", "floor", "deck", "pad" }

local function treadBelt()
    local t = treadmill()
    if not t then
        return nil
    end
    if t:IsA("BasePart") then
        return t
    end
    local best, bestScore
    for _, d in ipairs(t:GetDescendants()) do
        if d:IsA("BasePart") and not treadAvoided(d.Name) then
            local low = d.Name:lower()
            local bonus = 0
            for i = 1, #K.BELT_HINTS do
                if low:find(K.BELT_HINTS[i], 1, true) then
                    bonus = 100000
                    break
                end
            end
            local score = bonus + d.Size.X * d.Size.Z
            if not bestScore or score > bestScore then
                best, bestScore = d, score
            end
        end
    end
    return best or t.PrimaryPart
end

local function treadCFrame()
    local seat = treadSeat()
    if seat then
        return seat.CFrame
    end
    local part = treadBelt()
    return part and part.CFrame or nil
end

-- Seats weld on the server off a touch, so the prompt is tried first and parking the root
-- on the cushion is what closes it when there is no prompt.
local function sitOnTread(seat)
    local h, hrp = hum(), root()
    if not seat or not h or not hrp then
        return false
    end
    if h.Sit then
        return true
    end
    for _, pp in ipairs(promptsUnder(treadmill(), { "sit", "ride", "use", "start" })) do
        if promptInRange(pp) and firePrompt(pp) then
            task.wait(0.2)
            if h.Sit then
                return true
            end
        end
    end
    pcall(function() seat:Sit(h) end)
    local target = seat.CFrame * CFrame.new(0, 1.4, 0)
    for _ = 1, 24 do
        if h.Sit then
            return true
        end
        pcall(function() hrp.CFrame = target end)
        RunService.Heartbeat:Wait()
    end
    return h.Sit == true
end
local function upgradeTreadmill()
    local n = 0
    if callNet(R.TreadRaise) then
        n = n + 1
    end
    local t = treadmill()
    for _, pp in ipairs(promptsUnder(t, { "upgrade", "level", "tier", "buy" })) do
        if promptInRange(pp) and firePrompt(pp) then
            n = n + 1
            task.wait(0.06)
        end
    end
    return n
end

spawnLoop(function()
    local lastUp = 0
    while alive() do
        RunService.Heartbeat:Wait()
        if State.AutoTreadmill and charReady() and not M.travelling then
            local seat = treadSeat()
            local cf = treadCFrame()
            local hrp, h = root(), hum()
            if cf and hrp and h then
                local flat = Vector3.new(cf.Position.X - hrp.Position.X, 0, cf.Position.Z - hrp.Position.Z)
                if flat.Magnitude > 12 and not h.Sit then
                    travelTo(cf.Position, 6, function()
                        return alive() and State.AutoTreadmill
                    end)
                elseif seat then
                    if not h.Sit then
                        sitOnTread(seat)
                    end
                    if State.AutoUpgradeTreadmill and tick() - lastUp > 4 then
                        lastUp = tick()
                        pcall(upgradeTreadmill)
                    end
                else
                    local dir = cf.LookVector
                    local ahead = (hrp.Position - cf.Position):Dot(dir)
                    if ahead > 6 then
                        dir = -dir
                    end
                    pcall(function() h.WalkSpeed = math.max(h.WalkSpeed, 60) end)
                    h:Move(Vector3.new(dir.X, 0, dir.Z).Unit, false)
                    if State.AutoUpgradeTreadmill and tick() - lastUp > 4 then
                        lastUp = tick()
                        pcall(upgradeTreadmill)
                    end
                end
            end
        end
    end
end)

-- // Server hop. HttpGet to games.roblox.com is refused by a lot of executors, so the
-- executor request function is tried first and HttpGet is only the fallback.
M.rawRequest = (syn and syn.request) or (http and http.request) or http_request or request

local function fetch(url)
    if type(M.rawRequest) == "function" then
        local ok, res = pcall(M.rawRequest, { Url = url, Method = "GET" })
        if ok and type(res) == "table" and type(res.Body) == "string" then
            return res.Body
        end
    end
    local ok, body = pcall(function() return game:HttpGet(url) end)
    if ok and type(body) == "string" then
        return body
    end
    return nil
end

K.HOP_FILE = "ShardHopCount.txt"

local function loadHops()
    if isfile and readfile and isfile(K.HOP_FILE) then
        local ok, v = pcall(readfile, K.HOP_FILE)
        if ok then
            return tonumber(v) or 0
        end
    end
    return tonumber(getgenv and getgenv().__shardHops) or 0
end

local function saveHops(n)
    if writefile then
        pcall(writefile, K.HOP_FILE, tostring(n))
    end
    if getgenv then
        getgenv().__shardHops = n
    end
end
State.HopCount = loadHops()

M.visited = {}
M.hopping = false

local function serverList()
    local url = "https://games.roblox.com/v1/games/" .. game.PlaceId
        .. "/servers/Public?sortOrder=Desc&excludeFullGames=true&limit=100"
    local body = fetch(url)
    if not body then
        return {}
    end
    local ok, data = pcall(function() return HttpService:JSONDecode(body) end)
    if not ok or type(data) ~= "table" or type(data.data) ~= "table" then
        return {}
    end
    local out = {}
    for _, s in ipairs(data.data) do
        if s.id and s.id ~= game.JobId and not M.visited[s.id]
            and (s.playing or 0) < (s.maxPlayers or 30) then
            out[#out + 1] = s.id
        end
    end
    for i = #out, 2, -1 do
        local j = math.random(i)
        out[i], out[j] = out[j], out[i]
    end
    return out
end

local function serverHop()
    if M.hopping then
        return false
    end
    M.hopping = true
    local list = serverList()
    for _, id in ipairs(list) do
        M.visited[id] = true
        State.HopCount = State.HopCount + 1
        saveHops(State.HopCount)
        local ok = pcall(function()
            TeleportService:TeleportToPlaceInstance(game.PlaceId, id, player)
        end)
        if ok then
            task.wait(6)
            M.hopping = false
            return true
        end
    end
    pcall(function() TeleportService:Teleport(game.PlaceId, player) end)
    M.hopping = false
    return #list > 0
end
spawnLoop(function()
    while alive() do
        if State.AutoHop then
            if State.HopCount >= State.MaxHops then
                State.AutoHop = false
                setToggle("AutoHop", false)
                notify("Server Hop", "Hop limit reached.")
            else
                pcall(serverHop)
            end
        end
        task.wait(math.clamp(State.HopDelay, 5, 300))
    end
end)

local function spawnPoint()
    local sp = player.RespawnLocation
    if sp and sp:IsA("BasePart") then
        return sp.CFrame.Position + Vector3.new(0, 4, 0)
    end
    for _, d in ipairs(Workspace:GetDescendants()) do
        if d:IsA("SpawnLocation") then
            return d.Position + Vector3.new(0, 4, 0)
        end
    end
    local pv = plotPivot()
    return pv and pv.Position or nil
end

local function recallToSpawn()
    local p = spawnPoint()
    if not p then
        return false
    end
    return travelTo(p, 8)
end

local function unload()
    State.Unloaded = true
    State.AutoSteal = false
    State.AutoHop = false
    State.AutoHatch = false
    State.EggESP = false
    State.AutoEquipBest = false
    State.AutoSell = false
    State.AutoSellEggs = false
    State.AutoClaim = false
    State.AutoUpgrade = false
    State.AutoTreadmill = false
    State.AutoUpgradeTreadmill = false
    State.FpsBoost = false
    cancelTravel()
    fpsOff()
    clearESP()
    for _, c in ipairs(Conns) do
        pcall(function() c:Disconnect() end)
    end
    Conns = {}
    for _, th in ipairs(Threads) do
        pcall(task.cancel, th)
    end
    Threads = {}
    releaseRoot()
end

-- WindUI hands a multi dropdown back as an array, or as a map of name to true, or as
-- a bare string when only one is picked. All three are flattened to a plain list.
local function pickList(v)
    local out = {}
    if type(v) == "table" then
        for k, val in pairs(v) do
            if val == true then
                out[#out + 1] = k
            elseif type(val) == "string" then
                out[#out + 1] = val
            end
        end
    elseif type(v) == "string" then
        out[1] = v
    end
    return out
end

-- Minimising used to be a one way trip, so the reopen is driven three ways: the open
-- button, a keybind, and whichever restore method this WindUI build exposes.
local function showWindow()
    for _, m in ipairs({ "Open", "Maximize", "Unminimize", "Show" }) do
        if pcall(function() T.Window[m](T.Window) end) then
            return true
        end
    end
    return pcall(function() T.Window:Minimize(false) end)
end

    if not gp and input.KeyCode == Enum.KeyCode.RightControl then
